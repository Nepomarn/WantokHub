// ======================================== // WANTOKHUB - MAIN JAVASCRIPT // ======================================== // MOBILE MENU const menuBtn = document.getElementById("menuBtn"); const navLinks = document.getElementById("navLinks"); menuBtn.addEventListener("click", () => { navLinks.classList.toggle("active"); }); document.querySelectorAll(".nav-links a").forEach(link => { link.addEventListener("click", () => { navLinks.classList.remove("active"); }); }); // ======================================== // SEARCH SERVICES // ======================================== const searchInput = document.getElementById("siteSearch"); const searchBtn = document.getElementById("searchBtn"); const cards = document.querySelectorAll(".service-card"); const noResults = document.getElementById("noResults"); function searchServices() { const query = searchInput.value.trim().toLowerCase(); let found = 0; cards.forEach(card => { const content = ( card.innerText + " " + (card.dataset.search || "") ).toLowerCase(); const match = content.includes(query); card.style.display = match ? "" : "none"; if (match) found++; }); noResults.style.display = found === 0 ? "block" : "none"; document.getElementById("services").scrollIntoView({ behavior: "smooth" }); } searchBtn.addEventListener("click", searchServices); searchInput.addEventListener("keydown", event => { if (event.key === "Enter") { searchServices(); } }); // ======================================== // OPPORTUNITIES // ======================================== const latestContainer = document.getElementById("latestOpportunities"); // CREATE HTML ELEMENTS SAFELY function createElement(tag, className, text) { const element = document.createElement(tag); if (className) { element.className = className; } if (text !== undefined) { element.textContent = text; } return element; } // FORMAT DEADLINE function formatDate(dateValue) { if (!dateValue) return "Not specified"; const parts = dateValue.split("-"); if (parts.length !== 3) return dateValue; const date = new Date( Number(parts[0]), Number(parts[1]) - 1, Number(parts[2]) ); if (Number.isNaN(date.getTime())) { return dateValue; } return date.toLocaleDateString("en-PG", { day: "numeric", month: "short", year: "numeric" }); } // DEADLINE STATUS function getDeadlineStatus(deadlineValue) { if (!deadlineValue) return null; const parts = deadlineValue.split("-"); if (parts.length !== 3) return null; const deadline = new Date( Number(parts[0]), Number(parts[1]) - 1, Number(parts[2]) ); if (Number.isNaN(deadline.getTime())) return null; const today = new Date(); today.setHours(0, 0, 0, 0); deadline.setHours(0, 0, 0, 0); const daysLeft = Math.round( (deadline - today) / 86400000 ); if (daysLeft < 0) { return { label: "Deadline passed", className: "deadline-badge closed" }; } if (daysLeft === 0) { return { label: "Closing today", className: "deadline-badge today" }; } if (daysLeft <= 7) { return { label: daysLeft + " days left", className: "deadline-badge today" }; } return { label: "Open for applications", className: "deadline-badge normal" }; } // DISPLAY OPPORTUNITIES function displayOpportunities(opportunities) { latestContainer.replaceChildren(); if (!opportunities.length) { const message = createElement( "div", "opportunity-status" ); message.style.gridColumn = "1 / -1"; message.appendChild(createElement( "strong", "", "No current opportunities" )); message.appendChild(document.createTextNode( "New jobs, scholarships and other opportunities will appear here when published." )); latestContainer.appendChild(message); return; } // SHOW SIX LATEST OPPORTUNITIES opportunities.slice(0, 6).forEach(item => { const card = createElement( "article", "opportunity-card" ); // OPPORTUNITY TYPE card.appendChild(createElement( "span", "opportunity-type", item.type || "Opportunity" )); // DEADLINE BADGE const deadlineStatus = getDeadlineStatus(item.deadline); if (deadlineStatus) { card.appendChild(createElement( "span", deadlineStatus.className, deadlineStatus.label )); } // TITLE card.appendChild(createElement( "h3", "", item.title || "Untitled opportunity" )); // ORGANISATION card.appendChild(createElement( "p", "opportunity-organisation", item.organisation || "Organisation not specified" )); // INFORMATION const meta = createElement( "div", "opportunity-meta" ); meta.appendChild(createElement( "span", "", "📍 " + (item.province || "PNG") )); meta.appendChild(createElement( "span", "", "📅 Deadline: " + formatDate(item.deadline) )); card.appendChild(meta); // VIEW DETAILS const link = createElement( "a", "opportunity-card-link", "View opportunity details →" ); link.href = "/opportunities.html"; card.appendChild(link); latestContainer.appendChild(card); }); } // ======================================== // LOAD OPPORTUNITIES FROM API // ======================================== async function loadLatestOpportunities() { latestContainer.replaceChildren(); const loading = createElement( "div", "opportunity-status" ); loading.style.gridColumn = "1 / -1"; loading.appendChild(createElement( "strong", "", "Loading opportunities..." )); loading.appendChild(document.createTextNode( "Please wait while we retrieve the latest listings." )); latestContainer.appendChild(loading); try { const response = await fetch( "/api/opportunities", { method: "GET", headers: { "Accept": "application/json" }, cache: "no-store" } ); if (!response.ok) { throw new Error( "API returned status " + response.status ); } const data = await response.json(); if ( !data.success || !Array.isArray(data.opportunities) ) { throw new Error( "Invalid opportunities response." ); } displayOpportunities(data.opportunities); } catch (error) { console.error( "WantokHub opportunity error:", error ); latestContainer.replaceChildren(); const message = createElement( "div", "opportunity-status" ); message.style.gridColumn = "1 / -1"; message.appendChild(createElement( "strong", "", "Unable to load opportunities" )); message.appendChild(document.createTextNode( "Please try again later or visit the full opportunities page." )); latestContainer.appendChild(message); } } // ======================================== // CURRENT YEAR // ======================================== document.getElementById("year").textContent = new Date().getFullYear(); // ======================================== // START WEBSITE // ======================================== loadLatestOpportunities();
